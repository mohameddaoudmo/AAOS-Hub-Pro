الدليل الشامل لبناء خدمة نظام مخصصة
Custom System Service في AOSP

البلوك 9: اليوم 17-18 — من الصفر إلى الإنتاج
Android 13 / 14 — AOSP
١ معمارية خدمات النظام المخصصة

لماذا نبني خدمة نظام مخصصة بدلاً من Bound Service عادي؟

عند العمل على منصة AOSP (Android Open Source Project)، قد تحتاج إلى توفير واجهة برمجية (API) تتيح للتطبيقات الوصول إلى موارد نظامية أو عتاد (Hardware) لا يمكن الوصول إليه عبر واجهات Android العامة. في هذه الحالة، الحل الأمثل هو بناء Custom System Service تعمل ضمن عملية system_server وليس كـ Service عادي داخل تطبيق.

🔷 الفرق الجوهري

الـ Bound Service العادي يعمل ضمن Application Process ويُقتل عندما يُغلق التطبيق. أما الـ System Service فتعمل ضمن system_server وتبقى حية طوال عمر النظام، مما يوفر:

  • وصولاً مميزاً (Privileged Access) إلى HAL و Kernel Drivers.
  • استمرارية (Persistence) بغض النظر عن دورة حياة التطبيقات.
  • حماية عبر SELinux على مستوى النظام بالكامل.

الأعمدة الخمسة لإضافة خدمة نظام

الخطوة ٢.١: إنشاء ملف AIDL

يبدأ كل شيء بتعريف الواجهة عبر AIDL (Android Interface Definition Language). هذا الملف هو العقد الرسمي بين العميل (التطبيق) والخادم (System Service). يجب وضعه في المسار الصحيح ضمن frameworks/base/ ليتم بناؤه كجزء من الـ SDK الداخلي.

frameworks/base/core/java/android/os/ICustomHardwareService.aidl
// ICustomHardwareService.aidl
// واجهة AIDL للوصول إلى العتاد المخصص عبر System Service

package android.os;

// واجهة Binder IPC لتفعيل إجراءات العتاد المخصص
interface ICustomHardwareService {

    // تفعيل إجراء على العتاد وإرجاع حالة التنفيذ
    boolean triggerHardwareAction(int actionId);

    // قراءة قيمة استشعار (Sensor) مخصص
    int readSensorValue(int sensorType);

    // تعيين معامل تشغيلي للعتاد
    void setHardwareParameter(in String key, int value);

    // الحصول على إصدار البرنامج الثابت للعتاد
    String getFirmwareVersion();
}
💡 نصيحة هامة

يجب التأكد من أن الـ Package Name يبدأ بـ android.os أو android.app ليتم تضمينه تلقائياً في الـ Stubs التي يولدها AIDL Compiler ضمن بنية الـ Framework.

الخطوة ٢.٢: إنشاء الـ Manager Wrapper

الـ CustomHardwareManager هو الغلاف البرمجي (Proxy Wrapper) الذي تتعامل معه التطبيقات. يخفي هذا المدير تعقيدات استدعاءات Binder المباشرة ويوفر واجهة برمجية نظيفة وآمنة. يعمل كنمط Manager Pattern المعتاد في Android (مثل LocationManager، SensorManager).

frameworks/base/core/java/android/app/CustomHardwareManager.java
// CustomHardwareManager.java
// الغلاف البرمجي العام (Public API) للتطبيقات للوصول إلى خدمة العتاد المخصص

package android.app;

import android.annotation.SystemApi;
import android.content.Context;
import android.os.ICustomHardwareService;
import android.os.RemoteException;
import android.util.Log;

/**
 * CustomHardwareManager - مدير الوصول إلى العتاد المخصص
 *
 * يتم إنشاء هذا الكائن عبر SystemServiceRegistry ويتم تمريره
 * للتطبيقات عند استدعاء context.getSystemService().
 */
public class CustomHardwareManager {

    private static final String TAG = "CustomHardwareManager";

    // مرجع إلى الـ Binder Interface (AIDL Stub Proxy)
    private final ICustomHardwareService mService;

    // يتم استدعاء هذا المنشئ داخلياً من SystemServiceRegistry
    CustomHardwareManager(Context context, ICustomHardwareService service) {
        mService = service;
    }

    /**
     * تفعيل إجراء على العتاد المخصص.
     *
     * @param actionId معرف الإجراء المراد تنفيذه
     * @return true إذا نجح التنفيذ، false خلاف ذلك
     */
    public boolean triggerHardwareAction(int actionId) {
        try {
            return mService.triggerHardwareAction(actionId);
        } catch (RemoteException e) {
            Log.e(TAG, "فشل استدعاء triggerHardwareAction عبر Binder", e);
            return false;
        }
    }

    /**
     * قراءة قيمة استشعار مخصص.
     */
    public int readSensorValue(int sensorType) {
        try {
            return mService.readSensorValue(sensorType);
        } catch (RemoteException e) {
            Log.e(TAG, "فشل قراءة قيمة الاستشعار", e);
            return -1;
        }
    }

    /**
     * تعيين معامل تشغيلي للعتاد.
     */
    public void setHardwareParameter(String key, int value) {
        try {
            mService.setHardwareParameter(key, value);
        } catch (RemoteException e) {
            Log.e(TAG, "فشل تعيين المعامل التشغيلي", e);
        }
    }

    /**
     * الحصول على إصدار البرنامج الثابت للعتاد.
     */
    public String getFirmwareVersion() {
        try {
            return mService.getFirmwareVersion();
        } catch (RemoteException e) {
            Log.e(TAG, "فشل الحصول على إصدار البرنامج الثابت", e);
            return "unknown";
        }
    }
}

⚠️ تحذير أمني

يجب أن يكون اسم الخدمة (مثل "custom_hardware") فريداً عالمياً ولا يتعارض مع أي خدمة نظام أخرى. يُفضل استخدام بادئة تحدد الشركة المصنعة (OEM) لتجنب التعارضات في التحديثات المستقبلية.

٣ برمجة الخدمة داخل الخادم

الخطوة ٣.١: تنفيذ SystemService

الآن نصل إلى القلب النابض للعملية: تنفيذ الخدمة الفعلية التي تعمل داخل عملية system_server. يجب أن يمتد هذا الصنف من SystemService ويُسجّل نفسه كـ Binder Service عبر publishBinderService().

frameworks/base/services/core/java/com/android/server/CustomHardwareService.java
// CustomHardwareService.java
// التنفيذ الداخلي لخدمة العتاد المخصص داخل system_server

package com.android.server;

import android.app.CustomHardwareManager;
import android.content.Context;
import android.os.Binder;
import android.os.ICustomHardwareService;
import android.os.RemoteException;
import android.util.Log;

import com.android.server.SystemService;

/**
 * CustomHardwareService - خدمة النظام المخصصة للعتاد
 *
 * تعمل هذه الخدمة داخل عملية system_server وتتولى:
 * 1. التسجيل كـ Binder Service عبر ServiceManager
 * 2. التحقق من صلاحيات التطبيقات (Permission Checks)
 * 3. التواصل مع HAL أو Kernel Driver للوصول الفعلي إلى العتاد
 */
public class CustomHardwareService extends SystemService {

    private static final String TAG = "CustomHardwareService";
    private static final String PERMISSION_ACCESS_CUSTOM_HW
            = "android.permission.ACCESS_CUSTOM_HARDWARE";

    private final Context mContext;

    // Binder Stub - التنفيذ الفعلي لواجهة AIDL
    private final ICustomHardwareService.Stub mBinder = new ICustomHardwareService.Stub() {

        @Override
        public boolean triggerHardwareAction(int actionId) throws RemoteException {
            enforceCustomHardwarePermission();
            Log.i(TAG, "تفعيل الإجراء: " + actionId);

            // TODO: استدعاء HAL أو Kernel Driver هنا
            // مثال: mHal.triggerAction(actionId);

            return true;
        }

        @Override
        public int readSensorValue(int sensorType) throws RemoteException {
            enforceCustomHardwarePermission();
            Log.i(TAG, "قراءة الاستشعار: " + sensorType);

            // TODO: قراءة القيمة من العتاد
            return 42; // قيمة تجريبية
        }

        @Override
        public void setHardwareParameter(String key, int value) throws RemoteException {
            enforceCustomHardwarePermission();
            Log.i(TAG, "تعيين المعامل: " + key + " = " + value);

            // TODO: تطبيق المعامل على العتاد
        }

        @Override
        public String getFirmwareVersion() throws RemoteException {
            enforceCustomHardwarePermission();

            // TODO: قراءة الإصدار من العتاد
            return "v1.2.3-custom";
        }
    };

    // ───────────────────────────────────────────────
    // منشئ الخدمة - يتم استدعاؤه من SystemServer
    // ───────────────────────────────────────────────
    public CustomHardwareService(Context context) {
        super(context);
        mContext = context;
    }

    // ───────────────────────────────────────────────
    // onStart() - نقطة البداية عند تشغيل الخدمة
    // ───────────────────────────────────────────────
    @Override
    public void onStart() {
        Log.i(TAG, "بدء تشغيل CustomHardwareService...");

        // نشر الـ Binder Service في ServiceManager
        // الاسم هنا يجب أن يطابق Context.CUSTOM_HARDWARE_SERVICE
        publishBinderService(Context.CUSTOM_HARDWARE_SERVICE, mBinder);

        Log.i(TAG, "CustomHardwareService تم تسجيله بنجاح في ServiceManager");
    }

    // ───────────────────────────────────────────────
    // التحقق من صلاحيات الوصول إلى العتاد المخصص
    // ───────────────────────────────────────────────
    private void enforceCustomHardwarePermission() {
        mContext.enforceCallingOrSelfPermission(
                PERMISSION_ACCESS_CUSTOM_HW,
                "يتطلب إذن ACCESS_CUSTOM_HARDWARE"
        );
    }
}
🔷 آلية Permission Check

تستخدم enforceCallingOrSelfPermission() للتحقق من أن التطبيق الذي يستدعي الخدمة يملك الإذن المطلوب. هذا الإذن يجب تعريفه في frameworks/base/core/res/AndroidManifest.xml كـ <permission> على مستوى النظام (signature أو system).

الخطوة ٣.٢: تعريف الإذن في AndroidManifest.xml

frameworks/base/core/res/AndroidManifest.xml
<!-- إذن الوصول إلى العتاد المخصص - يقتصر على تطبيقات النظام -->
<permission android:name="android.permission.ACCESS_CUSTOM_HARDWARE"
    android:label="@string/permlab_accessCustomHardware"
    android:description="@string/permdesc_accessCustomHardware"
    android:protectionLevel="signature|privileged" />
⚠️ مستوى الحماية (Protection Level)

استخدام signature|privileged يضمن أن فقط التطبيقات الموقعة بنفس شهادة النظام أو الموجودة في /system/priv-app يمكنها الحصول على هذا الإذن. لا تستخدم normal أبداً لخدمات النظام الحساسة.

٤ تسجيل الخدمة في الفريمورك

الخطوة ٤.١: تعديل SystemServer.java

ملف SystemServer.java هو نقطة انطلاق جميع خدمات النظام. يجب إضافة سطر لبدء خدمتنا ضمن الدالة الرئيسية startOtherServices() أو أي قسم مناسب حسب تبعيات الخدمة.

frameworks/base/services/java/com/android/server/SystemServer.java
private void startOtherServices(@NonNull TimingsTraceAndSlog t) {
    // ... الخدمات الأخرى ...

    // ───────────────────────────────────────────────
    // تشغيل خدمة العتاد المخصصة
    // ───────────────────────────────────────────────
    t.traceBegin("StartCustomHardwareService");
    try {
        mSystemServiceManager.startService(CustomHardwareService.class);
    } catch (Throwable e) {
        Slog.e("System", "فشل تشغيل CustomHardwareService", e);
    }
    t.traceEnd();

    // ... باقي الخدمات ...
}
💡 ترتيب التشغيل

ضع خدمتك في الترتيب المناسب. إذا كانت خدمتك تعتمد على ActivityManagerService أو PackageManagerService، تأكد من تشغيلها بعد هذه الخدمات. يمكنك أيضاً استخدام startService(..., /* dependencies */) إذا كنت تستخدم إصداراً يدعم ذلك.

الخطوة ٤.٢: تعديل SystemServiceRegistry.java

هذا الملف هو "الجسر" الذي يربط بين اسم الخدمة (مثل Context.CUSTOM_HARDWARE_SERVICE) والكائن الفعلي (CustomHardwareManager). يستخدم نمط CachedServiceFetcher لإعادة استخدام نفس كائن الـ Manager لكل سياق (Context).

frameworks/base/core/java/android/app/SystemServiceRegistry.java
final class SystemServiceRegistry {

    // ... التسجيلات الموجودة مسبقاً ...

    static {
        // ... الخدمات الأخرى ...

        // ───────────────────────────────────────────────
        // تسجيل خدمة العتاد المخصصة
        // ───────────────────────────────────────────────
        registerService(Context.CUSTOM_HARDWARE_SERVICE,
                CustomHardwareManager.class,
                new CachedServiceFetcher<CustomHardwareManager>() {
            @Override
            public CustomHardwareManager createService(ContextImpl ctx)
                    throws ServiceNotFoundException {

                // الحصول على الـ Binder Interface من ServiceManager
                IBinder b = ServiceManager.getServiceOrThrow(
                        Context.CUSTOM_HARDWARE_SERVICE);

                // إنشاء الـ Manager وتمرير الـ Proxy
                ICustomHardwareService service =
                        ICustomHardwareService.Stub.asInterface(b);

                return new CustomHardwareManager(ctx, service);
            }
        });

        // ... باقي التسجيلات ...
    }
}

الخطوة ٥.١: إضافة الخدمة إلى service_contexts

ملف service_contexts يُخبر servicemanager بأنواع SELinux المرتبطة بكل خدمة Binder. بدون هذا التسجيل، سيرفض SELinux الوصول إلى الخدمة.

frameworks/base/core/java/android/content/Context.java

(أو device-specific: device/<vendor>/<board>/sepolicy/vendor/service_contexts)

# service_contexts - تعيين أنواع SELinux لخدمات Binder
# يجب إضافة السطر التالي:

custom_hardware    u:object_r:custom_hardware_service:s0

الخطوة ٥.٢: تعريف النوع (Type) في SELinux

device/<vendor>/<board>/sepolicy/vendor/service.te
# تعريف نوع الخدمة المخصصة
type custom_hardware_service, service_manager_type;

الخطوة ٥.٣: منح الصلاحيات (Permissions)

device/<vendor>/<board>/sepolicy/vendor/system_server.te
# السماح لـ system_server بإضافة (add) وإيجاد (find) الخدمة
allow system_server custom_hardware_service:service_manager { add find };
device/<vendor>/<board>/sepolicy/vendor/app.te
# السماح للتطبيقات (المميزة) بالعثور على الخدمة
allow untrusted_app custom_hardware_service:service_manager find;
allow system_app custom_hardware_service:service_manager find;
⚠️ أخطاء SELinux الشائعة

إذا نسيت أي من الخطوات السابقة، ستواجه أحد الأخطاء التالية في logcat:

  • avc: denied { add } — system_server لا يملك صلاحية إضافة الخدمة.
  • avc: denied { find } — التطبيق لا يملك صلاحية العثور على الخدمة.
  • ServiceManager: Permission failure — الخدمة غير مسجلة في service_contexts.

الخطوة ٥.٤: تحديث Android.bp

تأكد من تضمين ملفات AIDL في قواعد البناء. في Android 13/14، يتم ذلك عادةً تلقائياً إذا كان الملف في المسار الصحيح ضمن frameworks/base/، لكن قد تحتاج إلى تحديث Android.bp إذا كنت تستخدم مساراً مخصصاً.

frameworks/base/Android.bp (أو Android.mk للإصدارات القديمة)

java_defaults {
    name: "framework-defaults",
    srcs: [
        "core/java/android/os/ICustomHardwareService.aidl",
        // ... الملفات الأخرى ...
    ],
}

الخطوة ٥.٥: أوامر البناء

# 1. تحديث واجهات برمجة التطبيقات (API Stubs)
$ make update-api

# 2. بناء الـ Framework بأكمله
$ make framework -j$(nproc)

# 3. بناء خدمات النظام
$ make services -j$(nproc)

# 4. بناء SELinux Policies
$ make sepolicy -j$(nproc)

# 5. تثبيت النتيجة على الجهاز (اختياري)
$ adb root && adb remount
$ adb sync
$ adb reboot
💡 تلميح للبناء السريع

للتطوير السريع، يمكنك بناء فقط framework.jar و services.jar ثم دفعهما عبر adb sync دون إعادة بناء النظام بالكامل. لكن تذكر أن تغييرات AIDL تتطلب إعادة بناء كاملة.

٦ استدعاء الخدمة من التطبيقات

كيفية استخدام الخدمة في تطبيق Android

بعد بناء النظام وإعادة تشغيله، يمكن لأي تطبيق يملك الإذن android.permission.ACCESS_CUSTOM_HARDWARE الوصول إلى الخدمة بنفس طريقة الوصول إلى أي خدمة نظام أخرى.

MyApplication.java (تطبيق النظام في /system/priv-app)
package com.example.myapp;

import android.app.Activity;
import android.app.CustomHardwareManager;
import android.content.Context;
import android.os.Bundle;
import android.util.Log;

public class MainActivity extends Activity {

    private static final String TAG = "MyApp";

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        // ───────────────────────────────────────────────
        // الحصول على CustomHardwareManager عبر Context
        // ───────────────────────────────────────────────
        CustomHardwareManager manager =
                (CustomHardwareManager) getSystemService(
                        Context.CUSTOM_HARDWARE_SERVICE);

        if (manager != null) {
            // تفعيل إجراء على العتاد
            boolean success = manager.triggerHardwareAction(1);
            Log.d(TAG, "نتيجة التفعيل: " + success);

            // قراءة قيمة استشعار
            int sensorValue = manager.readSensorValue(0);
            Log.d(TAG, "قيمة الاستشعار: " + sensorValue);

            // تعيين معامل
            manager.setHardwareParameter("brightness", 80);

            // الحصول على إصدار البرنامج الثابت
            String version = manager.getFirmwareVersion();
            Log.d(TAG, "إصدار البرنامج الثابت: " + version);
        } else {
            Log.e(TAG, "CustomHardwareManager غير متوفر!");
        }
    }
}

    <!-- طلب إذن الوصول إلى العتاد المخصص -->
    <uses-permission android:name="android.permission.ACCESS_CUSTOM_HARDWARE" />

    <application ...>
        <activity android:name=".MainActivity">
            <intent-filter>
                <action android:name="android.intent.action.MAIN" />
                <category android:name="android.intent.category.LAUNCHER" />
            </intent-filter>
        </activity>
    </application>

</manifest>
٧ التشخيص واستكشاف الأخطاء

أوامر التشخيص الأساسية

# التحقق من تسجيل الخدمة في ServiceManager
$ adb shell service list | grep custom_hardware

# مراقبة سجلات system_server عند بدء التشغيل
$ adb logcat -s CustomHardwareService:SystemServer *:S

# فحص أخطاء SELinux (AVC Denials)
$ adb shell dmesg | grep avc
$ adb shell logcat -b events | grep avc

# التحقق من أن الـ Context Constant متاح
$ adb shell cmd activity service custom_hardware

# إعادة تحميل SELinux Policies (للتطوير فقط)
$ adb shell setenforce 0  # وضع Permissive (غير آمن للإنتاج)

جدول الأخطاء الشائعة والحلول

الخطأ السبب المحتمل الحل
ServiceNotFoundException الخدمة غير مسجلة في SystemServiceRegistry تحقق من تسجيل CachedServiceFetcher
SecurityException التطبيق لا يملك الإذن المطلوب تأكد من protectionLevel وموقع التطبيق في /system/priv-app
avc: denied { add } SELinux يرفض إضافة الخدمة أضف القاعدة في system_server.te
ServiceManager: Permission failure الخدمة غير موجودة في service_contexts أضف التعيين في service_contexts
الـ Manager يُرجع null الـ AIDL لم يُبنَ بشكل صحيح شغّل make update-api ثم أعد البناء
🔷 أدوات متقدمة

لتصحيح أخطاء Binder IPC المعقدة، استخدم:

  • adb shell dumpsys binder — لعرض إحصائيات Binder.
  • adb shell lshal — للتحقق من HALs المسجلة (إذا كنت تستخدم HAL).
  • adb shell service check custom_hardware — للتحقق السريع من توفر الخدمة.

📝 خطوات البناء النهائية

  1. تعديل جميع الملفات المذكورة أعلاه.
  2. تشغيل make update-api لتحديث واجهات البرمجة.
  3. تشغيل make framework services sepolicy -j$(nproc).
  4. تثبيت النتيجة عبر adb sync وإعادة التشغيل.
  5. التحقق من adb shell service list | grep custom_hardware.
  6. اختبار التطبيق والتحقق من SELinux عبر adb shell dmesg | grep avc.
